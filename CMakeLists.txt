cmake_minimum_required(VERSION 3.22)
project(fontreal LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Widgets)
find_package(Freetype REQUIRED)
add_library(experiment src/experiment.cpp src/window.cpp)
target_include_directories(experiment PUBLIC src)
target_link_libraries(experiment PUBLIC Qt6::Widgets Freetype::Freetype)
target_compile_options(experiment PRIVATE -Wall -Wextra -Wpedantic)
add_executable(fontreal src/main.cpp)
target_link_libraries(fontreal PRIVATE experiment)
option(FONTREAL_KONSOLE_RGB "Build the stock Konsole RGB pilot" OFF)
if(FONTREAL_KONSOLE_RGB)
  find_package(Fontconfig REQUIRED)
  add_library(fontreal-qt-rgb SHARED
    src/fontreal_qt_rgb.cpp src/fontreal_v3_outline.cpp src/fontreal_v3_coverage.cpp)
  target_link_libraries(fontreal-qt-rgb PRIVATE
    Qt6::Widgets Freetype::Freetype Fontconfig::Fontconfig ${CMAKE_DL_LIBS})
  target_compile_options(fontreal-qt-rgb PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 6.8 QUIET COMPONENTS Quick QuickControls2 DBus GuiPrivate)
  if(TARGET Qt6::Quick AND TARGET Qt6::QuickControls2 AND TARGET Qt6::DBus AND TARGET Qt6::GuiPrivate)
    add_executable(qt-kde-probe tests/kde_text_probe.cpp)
    target_link_libraries(qt-kde-probe PRIVATE Qt6::Widgets Qt6::Quick Qt6::QuickControls2 Qt6::DBus Qt6::GuiPrivate Freetype::Freetype)
    target_compile_definitions(qt-kde-probe PRIVATE FONTREAL_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_options(qt-kde-probe PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_executable(qt-native-probe tests/qt_native_probe.cpp)
  target_link_libraries(qt-native-probe PRIVATE Qt6::Widgets Freetype::Freetype)
  target_compile_options(qt-native-probe PRIVATE -Wall -Wextra -Wpedantic)
  add_library(qt-native-trace SHARED tests/qt_native_trace.cpp)
  target_link_libraries(qt-native-trace PRIVATE Freetype::Freetype ${CMAKE_DL_LIBS})
  add_executable(experiment-test tests/experiment_test.cpp)
  target_link_libraries(experiment-test PRIVATE experiment)
  add_executable(v3-coverage-test tests/v3_coverage_test.cpp src/fontreal_v3_coverage.cpp)
  target_include_directories(v3-coverage-test PRIVATE src)
  target_compile_options(v3-coverage-test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME v3-coverage COMMAND v3-coverage-test)
  add_test(NAME experiment-core COMMAND experiment-test $<TARGET_FILE:fontreal> core)
  add_test(NAME experiment-view-100 COMMAND experiment-test $<TARGET_FILE:fontreal> view)
  add_test(NAME experiment-view-125 COMMAND experiment-test $<TARGET_FILE:fontreal> view)
  set_tests_properties(experiment-core experiment-view-100 PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=1")
  set_tests_properties(experiment-view-125 PROPERTIES
    ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=1.25")
endif()
